% pdf-preamble.tex
%
% LaTeX setup for the PDF output of the exercise sheets.
%
% indent_filter.lua turns the two block types of the sheets into the two
% mdframed environments at the end of this file. Both indent their content by
% one list level (\leftmargini), i.e. align it with the text of the subtasks
% a), b), ...:
% - solution-block (::: {.sol}): with a vertical line on the left that marks
%   the solution
% - indent-block (::: {.indent}): indentation only

\usepackage{mdframed}
\usepackage{fancyvrb}
\usepackage{needspace}
\usepackage{etoolbox}
% Jupyter's R kernel renders character-vector outputs as inline lists,
% which pandoc writes as enumerate*; enumitem's [inline] option defines
% that environment.
\usepackage[inline]{enumitem}
% Avoid ugly page breaks (see also pagebreak_filter.lua for headings):
% never leave a single line of a paragraph alone at the top or bottom of
% a page, and don't start a code block in the last lines of a page.
\widowpenalty=10000
\clubpenalty=10000
\displaywidowpenalty=10000
\BeforeBeginEnvironment{Shaded}{\needspace{4\baselineskip}}
% Keep headings with the first lines of their content. This also covers
% the R/Python tab titles: Quarto turns panel-tabsets into subsubsections
% after the Lua filters have run, so they can only be caught here.
% Exercise headings (subsection) are followed by a learning-goals box and
% need more room; tab titles (subsubsection) are followed by a code chunk.
\AddToHook{cmd/subsection/before}{\needspace{10\baselineskip}}
% tab titles are often followed by a caption line that itself requires
% 8 lines (see pagebreak_filter.lua), so they must request heading
% (incl. its surrounding skips, ~4-5 lines) + that
\AddToHook{cmd/subsubsection/before}{\needspace{14\baselineskip}}
% gap between the vertical line of a solution and its text
\newlength{\indentframegap}
\setlength{\indentframegap}{0.5em}
% Reduce code font size in PDF code blocks generated by Pandoc/Quarto.
\AtBeginDocument{\DefineVerbatimEnvironment{Highlighting}{Verbatim}{commandchars=\\\{\},fontsize=\fontsize{9.6pt}{11pt}\selectfont}}
\mdfdefinestyle{sheetblock}{topline=false,bottomline=false,rightline=false,%
  innerrightmargin=0pt,innertopmargin=0pt,innerbottommargin=0pt}
\newmdenv[style=sheetblock,linewidth=1pt,linecolor=black,%
  leftmargin=\dimexpr\leftmargini-\indentframegap\relax,innerleftmargin=\indentframegap]{solution-block}
\newmdenv[style=sheetblock,linewidth=0pt,%
  leftmargin=\leftmargini,innerleftmargin=0pt]{indent-block}
